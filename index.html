function extractIntensityProfile(curve) { intensityProfile = []; if ( procCanvas.width !== currentImage.width || procCanvas.height !== currentImage.height ) { procCanvas.width = currentImage.width; procCanvas.height = currentImage.height; procCtx.drawImage(currentImage,0,0); } curve.forEach((p,index)=>{ const x = Math.round(p.x); const y = Math.round(p.y); if ( x < 0 || y < 0 || x >= currentImage.width || y >= currentImage.height ) return; const px = procCtx.getImageData(x,y,1,1).data; const lum = px[0]*0.299 + px[1]*0.587 + px[2]*0.114; intensityProfile.push({ x:p.x, y:p.y, luminance:lum, index:index }); }); }
